function cleanBodyHtml(html) { const decoded = decodeHtml(html || ""); const temp = document.createElement("div"); temp.innerHTML = decoded; // 중복 제목 제거: h2/h3/p 모두 검사 temp.querySelectorAll("h2, h3, p").forEach(function(el) { const trimmed = (el.textContent || "").trim(); if ( trimmed === "오늘의 노무 이슈" || trimmed === "오늘의 판례" || trimmed === "오늘의 노동법 공부" ) { el.remove(); } }); temp.querySelectorAll("a").forEach(function(a) { const text = document.createTextNode(a.textContent || ""); a.parentNode.replaceChild(text, a); }); temp.querySelectorAll("font").forEach(function(f) { const text = document.createTextNode(f.textContent || ""); f.parentNode.replaceChild(text, f); }); temp.querySelectorAll("p").forEach(function(p) { let text = p.textContent || ""; const trimmed = text.trim(); if (trimmed === "원문 보기" || trimmed === "원문보기" || trimmed.indexOf("원문:") === 0) { p.remove(); return; } text = removeLongUrls(text).trim(); const match = text.match(/^(\d+\.\s[^\.]+?)(\s+)(.+)$/); if (match) { const heading = match[1].trim(); const body = match[3].trim(); p.innerHTML = body ? "" + heading + "

" + body : "" + heading + ""; } else { p.textContent = text; } }); return temp.innerHTML; }
